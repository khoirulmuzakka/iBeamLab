cmake_minimum_required(VERSION 3.24)
project(iBeamLab VERSION 0.1.0 LANGUAGES CXX)

if(POLICY CMP0144)
    cmake_policy(SET CMP0144 NEW)
endif()

# Project configuration is defined here; cpp/ contains target definitions only.
option(BUILD_SHARED_LIBS "Build shared library" OFF)
option(IBEAMLAB_BUILD_TESTS "Build C++ and enabled Python tests" ON)
option(IBEAMLAB_CORE_BUILD_TESTS "Build core tests" ON)
option(IBEAMLAB_BUILD_PYTHON "Build Python bindings" OFF)
option(IBEAMLAB_BUILD_EXAMPLES "Build C++ examples" OFF)
option(IBEAMLAB_BUILD_BENCHMARKS "Build benchmarks" OFF)
option(IBEAMLAB_ENABLE_SIMNRA "Enable Windows SIMNRA COM backend" OFF)
option(IBEAMLAB_ENABLE_ONNX "Enable ONNX Runtime inference" OFF)
option(IBEAMLAB_SANITIZERS "Enable ASan and UBSan (GCC/Clang)" OFF)
option(IBEAMLAB_CHECK_SAVED_SIMNRA_SPECTRA "Require saved XNRA curves to match recalculation" OFF)
option(IBEAMLAB_REQUIRE_SIMNRA_TEST_RUNTIME "Fail instead of skipping if SIMNRA is unavailable" OFF)
set(ONNXRUNTIME_ROOT "" CACHE PATH "ONNX Runtime C/C++ SDK root")

set(IBEAMLAB_RUNTIME_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${IBEAMLAB_RUNTIME_DIRECTORY}/$<0:>")
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${IBEAMLAB_RUNTIME_DIRECTORY}/$<0:>")
set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib/$<0:>")
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
list(APPEND CMAKE_MODULE_PATH "${PROJECT_SOURCE_DIR}/cmake")

if(IBEAMLAB_BUILD_TESTS)
    enable_testing()
endif()
add_subdirectory(cpp)
