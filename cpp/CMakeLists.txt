cmake_minimum_required(VERSION 3.24)
project(iBeamLabCpp VERSION 0.1.0 LANGUAGES CXX)
if(POLICY CMP0144)
    cmake_policy(SET CMP0144 NEW)
endif()
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/bin/$<0:>")
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/bin/$<0:>")
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
list(APPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/cmake")
option(IBEAMLAB_BUILD_TESTS "Build C++ tests" ON)
option(IBEAMLAB_CORE_BUILD_TESTS "Build core tests" ON)
option(IBEAMLAB_ENABLE_SIMNRA "Enable Windows SIMNRA COM backend" OFF)
option(IBEAMLAB_ENABLE_ONNX "Enable ONNX Runtime inference" OFF)
option(IBEAMLAB_BUILD_PYTHON "Build independent Python bindings" OFF)
option(IBEAMLAB_BUILD_EXAMPLES "Build examples" OFF)
option(IBEAMLAB_BUILD_BENCHMARKS "Build benchmarks" OFF)
option(IBEAMLAB_SANITIZERS "Enable ASan and UBSan (GCC/Clang)" OFF)
option(BUILD_SHARED_LIBS "Build shared library" OFF)
option(IBEAMLAB_CHECK_SAVED_SIMNRA_SPECTRA "Require saved XNRA curves to match recalculation" OFF)
option(IBEAMLAB_REQUIRE_SIMNRA_TEST_RUNTIME "Fail instead of skipping if SIMNRA is unavailable" OFF)
set(ONNXRUNTIME_ROOT "" CACHE PATH "ONNX Runtime C/C++ SDK root")

add_library(ibeamlab
    src/core/sample.cpp src/core/parameter.cpp src/core/validation.cpp
    src/processing/rebin.cpp src/processing/pileup.cpp src/processing/transforms.cpp
    src/simulation/simulator.cpp src/simulation/simnra.cpp
    src/datasets/dataset.cpp src/generation/generator.cpp
    src/model/metadata.cpp src/model/package.cpp src/inference/session.cpp src/inference/models.cpp)
add_library(ibeamlab::ibeamlab ALIAS ibeamlab)
add_library(ibeamlab::core ALIAS ibeamlab)
target_compile_features(ibeamlab PUBLIC cxx_std_20)
set_target_properties(ibeamlab PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(ibeamlab PUBLIC
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
    $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}> PRIVATE src)
if(BUILD_SHARED_LIBS)
    target_compile_definitions(ibeamlab PUBLIC IBEAMLAB_SHARED)
    target_compile_definitions(ibeamlab PRIVATE IBEAMLAB_BUILDING_LIBRARY)
endif()
if(MSVC)
    target_compile_options(ibeamlab PRIVATE /W4 /permissive- /utf-8)
else()
    target_compile_options(ibeamlab PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(IBEAMLAB_SANITIZERS AND NOT MSVC)
    target_compile_options(ibeamlab PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(ibeamlab PUBLIC -fsanitize=address,undefined)
endif()
find_package(Threads REQUIRED)
target_link_libraries(ibeamlab PRIVATE Threads::Threads)
if(IBEAMLAB_ENABLE_SIMNRA)
    if(NOT WIN32)
        message(FATAL_ERROR "SIMNRA requires Windows")
    endif()
    target_compile_definitions(ibeamlab PRIVATE IBEAMLAB_HAS_SIMNRA)
    target_link_libraries(ibeamlab PRIVATE ole32 oleaut32)
endif()
if(IBEAMLAB_ENABLE_ONNX)
    find_path(ONNXRUNTIME_INCLUDE_DIR onnxruntime_cxx_api.h HINTS "${ONNXRUNTIME_ROOT}" PATH_SUFFIXES include build/native/include buildTransitive/native/include)
    find_library(ONNXRUNTIME_LIBRARY onnxruntime HINTS "${ONNXRUNTIME_ROOT}" PATH_SUFFIXES lib lib64 runtimes/win-x64/native)
    if(NOT ONNXRUNTIME_INCLUDE_DIR OR NOT ONNXRUNTIME_LIBRARY)
        message(FATAL_ERROR "Provide ONNXRUNTIME_ROOT pointing to an ONNX Runtime SDK")
    endif()
    target_include_directories(ibeamlab PRIVATE "${ONNXRUNTIME_INCLUDE_DIR}")
    find_package(ONNXRuntime REQUIRED)
    target_link_libraries(ibeamlab PRIVATE ONNXRuntime::Runtime)
    if(NOT BUILD_SHARED_LIBS)
        install(FILES "${ONNXRUNTIME_LIBRARY}" DESTINATION ${CMAKE_INSTALL_LIBDIR})
    endif()
    target_compile_definitions(ibeamlab PRIVATE IBEAMLAB_HAS_ONNX)
    if(WIN32)
        find_file(ONNXRUNTIME_DLL onnxruntime.dll HINTS "${ONNXRUNTIME_ROOT}" PATH_SUFFIXES bin lib runtimes/win-x64/native REQUIRED)
        configure_file("${ONNXRUNTIME_DLL}" "${CMAKE_CURRENT_BINARY_DIR}/bin/onnxruntime.dll" COPYONLY)
        install(FILES "${ONNXRUNTIME_DLL}" DESTINATION ${CMAKE_INSTALL_BINDIR})
    endif()
endif()

if(IBEAMLAB_BUILD_TESTS)
    enable_testing()
    if(IBEAMLAB_CORE_BUILD_TESTS)
        add_executable(ibeamlab_core_tests tests/core/core_tests.cpp)
        target_link_libraries(ibeamlab_core_tests PRIVATE ibeamlab)
        add_test(NAME core COMMAND ibeamlab_core_tests)
        set_tests_properties(core PROPERTIES LABELS "unit")
    endif()
    add_executable(ibeamlab_workflow_tests tests/workflow_tests.cpp)
    target_link_libraries(ibeamlab_workflow_tests PRIVATE ibeamlab)
    target_compile_definitions(ibeamlab_workflow_tests PRIVATE IBEAMLAB_FIXTURES="${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures")
    if(IBEAMLAB_ENABLE_ONNX)
        target_compile_definitions(ibeamlab_workflow_tests PRIVATE IBEAMLAB_TEST_ONNX)
    endif()
    foreach(suite processing generation packages)
        add_test(NAME ${suite} COMMAND ibeamlab_workflow_tests ${suite})
        set_tests_properties(${suite} PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}")
    endforeach()
    set_tests_properties(processing PROPERTIES LABELS "unit")
    set_tests_properties(generation packages PROPERTIES LABELS "integration")
    if(IBEAMLAB_CORE_BUILD_TESTS)
        set_tests_properties(core PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}")
    endif()
    add_test(NAME installed-consumer COMMAND ${CMAKE_COMMAND}
        -DBUILD_DIR=${CMAKE_CURRENT_BINARY_DIR} -DSOURCE_DIR=${CMAKE_CURRENT_SOURCE_DIR}
        -DCONFIG=$<CONFIG> -P ${CMAKE_CURRENT_SOURCE_DIR}/tests/installed.cmake)
    set_tests_properties(installed-consumer PROPERTIES LABELS "integration;deployment" RUN_SERIAL TRUE)
endif()
if(IBEAMLAB_BUILD_BENCHMARKS)
    add_executable(ibeamlab_benchmark benchmarks/processing.cpp)
    target_link_libraries(ibeamlab_benchmark PRIVATE ibeamlab)
    if(IBEAMLAB_ENABLE_ONNX)
        target_compile_definitions(ibeamlab_benchmark PRIVATE IBEAMLAB_BENCHMARK_ONNX IBEAMLAB_FIXTURES="${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures")
    endif()
    if(IBEAMLAB_BUILD_TESTS)
        add_test(NAME allocation-benchmark COMMAND ibeamlab_benchmark)
        set_tests_properties(allocation-benchmark PROPERTIES LABELS "performance")
    endif()
endif()
if(IBEAMLAB_BUILD_EXAMPLES)
    add_executable(ibeamlab_generate examples/generate.cpp)
    target_link_libraries(ibeamlab_generate PRIVATE ibeamlab)
endif()
if(IBEAMLAB_BUILD_PYTHON)
    find_package(Python COMPONENTS Interpreter Development.Module REQUIRED)
    execute_process(COMMAND "${Python_EXECUTABLE}" -m pybind11 --cmakedir OUTPUT_VARIABLE pybind11_DIR OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
    find_package(pybind11 CONFIG REQUIRED)
    pybind11_add_module(_ibeamlab python/module.cpp)
    target_link_libraries(_ibeamlab PRIVATE ibeamlab)
    target_include_directories(_ibeamlab PRIVATE src)
    install(TARGETS _ibeamlab LIBRARY DESTINATION ibeamlab RUNTIME DESTINATION ibeamlab)
    install(FILES python/ibeamlab/__init__.py python/ibeamlab/export.py DESTINATION ibeamlab)
    if(BUILD_SHARED_LIBS)
        install(FILES $<TARGET_FILE:ibeamlab> DESTINATION ibeamlab)
    endif()
    if(IBEAMLAB_ENABLE_ONNX AND WIN32)
        install(FILES "${ONNXRUNTIME_DLL}" DESTINATION ibeamlab)
    endif()
    if(IBEAMLAB_BUILD_TESTS)
        add_test(NAME python-api COMMAND "${Python_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/python_api.py" "$<TARGET_FILE_DIR:_ibeamlab>" "${IBEAMLAB_ENABLE_ONNX}")
        set_tests_properties(python-api PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}" LABELS "integration;python")
        if(IBEAMLAB_ENABLE_SIMNRA)
            file(GLOB IBEAMLAB_XNRA_REFERENCES "${CMAKE_CURRENT_SOURCE_DIR}/../xnra/*.xnra")
            if(IBEAMLAB_XNRA_REFERENCES)
                set(simnra_test_arguments)
                if(IBEAMLAB_CHECK_SAVED_SIMNRA_SPECTRA)
                    list(APPEND simnra_test_arguments --check-saved-spectra)
                endif()
                if(NOT IBEAMLAB_REQUIRE_SIMNRA_TEST_RUNTIME)
                    list(APPEND simnra_test_arguments --skip-unavailable)
                endif()
                add_test(NAME simnra-reference COMMAND "${Python_EXECUTABLE}"
                    "${CMAKE_CURRENT_SOURCE_DIR}/tests/simnra_xnra.py"
                    "$<TARGET_FILE_DIR:_ibeamlab>" ${IBEAMLAB_XNRA_REFERENCES}
                    --report "${CMAKE_CURRENT_BINARY_DIR}/simnra-xnra-results.json" ${simnra_test_arguments})
                set_tests_properties(simnra-reference PROPERTIES SKIP_RETURN_CODE 77 TIMEOUT 300 LABELS "integration;licensed;simnra" RUN_SERIAL TRUE)
            endif()
        endif()
    endif()
endif()
install(TARGETS ibeamlab EXPORT ibeamlab-targets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR} LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR} RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR} PATTERN .gitkeep EXCLUDE)
install(EXPORT ibeamlab-targets NAMESPACE ibeamlab:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/ibeamlab)
set(IBEAMLAB_CONFIG_NEEDS_ONNX OFF)
if(IBEAMLAB_ENABLE_ONNX AND NOT BUILD_SHARED_LIBS)
    set(IBEAMLAB_CONFIG_NEEDS_ONNX ON)
endif()
install(FILES cmake/FindONNXRuntime.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/ibeamlab)
configure_package_config_file(cmake/ibeamlab-config.cmake.in "${CMAKE_CURRENT_BINARY_DIR}/ibeamlab-config.cmake" INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/ibeamlab)
write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/ibeamlab-config-version.cmake" VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/ibeamlab-config.cmake" "${CMAKE_CURRENT_BINARY_DIR}/ibeamlab-config-version.cmake" DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/ibeamlab)
